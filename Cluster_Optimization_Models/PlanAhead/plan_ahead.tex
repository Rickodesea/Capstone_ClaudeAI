\section{Overview}

The plan-ahead model is a \textbf{Mixed-Integer Second-Order Cone Program (MISOCP)} solved
once per planning horizon. It answers:

\begin{quote}
\emph{Which machines should each tenant be assigned to for each planning period,
given their estimated resource demand, machine pool, and feedback from the previous horizon?}
\end{quote}

\subsection*{What the Model Decides}
\begin{enumerate}[label=(\arabic*)]
  \item Which additional machines (beyond the always-available pool) to activate.
  \item Which machines to assign \emph{exclusively} to exclusive tenants (fixed for the full horizon).
  \item Which machines to assign to \emph{shared} tenants per period (can change each period).
  \item How to mix high-usage and low-usage shared tenants on the same machines.
\end{enumerate}

\subsection*{Output}
An ordered list of planning periods. Each period has a list of \textbf{tenant groups}.
Each group specifies the tenant IDs and their assigned machine IDs for that period.
All tenants appear in every period. The Cluster Manager calls the Real-Time model
once per group per interval (multiple times per period), passing only the pre-filtered jobs and machines.

\subsection*{Two Solver Modes}
\begin{center}
\begin{tabular}{lll}
\toprule
Mode & \texttt{use\_socp} & Capacity constraint \\
\midrule
MILP  & \texttt{False} & Hard linear bound \\
MISOCP & \texttt{True} & Cantelli probabilistic bound (distribution-free) \\
\bottomrule
\end{tabular}
\end{center}

$\ufb_{ih}$ & Feedback-adjusted demand: $\ufb_{ih} = u_{ih}(1+\beta\bar{W}_i/\bar{W}_{\mathrm{ref}})$ \\

% ─────────────────────────────────────────────────────────────
\section{Constraints}

\subsection{Machine Activation}

\paragraph{C\textsubscript{aa} --- Always Available.}
Always-available machines are always active:
\begin{equation}\tag{C$_{\mathrm{aa}}$}
  z_{nh} = 1 \qquad \forall\; n \in \Ma,\ h \in \H
\end{equation}

\paragraph{C\textsubscript{act} --- Additional Machine Activation.}
Additional machines are active only if switched on:
\begin{equation}\tag{C$_{\mathrm{act}}$}
  z_{nh} \leq z^{\mathrm{on}}_n \qquad \forall\; n \in \Mb,\ h \in \H
\end{equation}

\paragraph{C\textsubscript{zact} --- Machine Active if Assigned.}
A machine must be active if any shared tenant is assigned to it:
\begin{equation}\tag{C$_{\mathrm{zact}}$}
  z_{nh} \geq y_{inh} \qquad \forall\; i \in \Ts,\ n \in \M,\ h \in \H
\end{equation}

\subsection{Exclusive Tenant Constraints}

\paragraph{C\textsubscript{excl1} --- At Most One Exclusive Per Machine.}
\begin{equation}\tag{C$_{\mathrm{excl1}}$}
  \sum_{i \in \Te} e_{in} \leq 1 \qquad \forall\; n \in \M
\end{equation}

\paragraph{C\textsubscript{excl2} --- Each Exclusive Must Be Assigned.}
\begin{equation}\tag{C$_{\mathrm{excl2}}$}
  \sum_{n \in \M} e_{in} \geq 1 \qquad \forall\; i \in \Te
\end{equation}

\paragraph{C\textsubscript{excl\_cap} --- Exclusive Capacity Coverage.}
Machines assigned exclusively to tenant $i$ must together cover its peak demand:
\begin{equation}\tag{C$_{\mathrm{excl\_cap}}$}
  \sum_{n \in \M} e_{in}\cdot C_n \geq \umax_i \qquad \forall\; i \in \Te
\end{equation}

\paragraph{C\textsubscript{sep} --- Exclusive-Shared Separation.}
A machine assigned exclusively to any exclusive tenant cannot be used by shared tenants:
\begin{equation}\tag{C$_{\mathrm{sep}}$}
  \sum_{i \in \Te} e_{in} + y_{jnh} \leq 1
  \qquad \forall\; j \in \Ts,\ n \in \M,\ h \in \H
\end{equation}

\subsection{Shared Tenant Constraints}

\paragraph{C\textsubscript{share} --- Every Shared Tenant Assigned Minimum Machines Every Period.}
Each shared tenant must receive at least $m_{\min}$ machines per period
(default $m_{\min}=2$; configurable via \texttt{min\_machines\_per\_tenant}):
\begin{equation}\tag{C$_{\mathrm{share}}$}
  \sum_{n \in \M} y_{inh} \geq m_{\min} \qquad \forall\; i \in \Ts,\ h \in \H
\end{equation}
Setting $m_{\min}\geq 2$ guarantees resilience: tenants stay scheduled even when one machine fails or spikes.

\paragraph{C\textsubscript{1a} --- Capacity with Cantelli Safety Buffer.}
\begin{equation}\tag{C$_{1a}$}
  \sum_{i \in \Ts} f_{inh} + \kappa\cdot t_{nh}
  \;\leq\; \Ceff_n \cdot z_{nh}
  \qquad \forall\; n \in \M,\ h \in \H
\end{equation}
In MILP mode: same constraint without the $\kappa\cdot t_{nh}$ term (and $t_{nh}$ is absent).

\paragraph{C\textsubscript{1b} --- Cantelli Cone (MISOCP only).}
\begin{equation}\tag{C$_{1b}$}
  \sum_{i \in \Ts} \sigma^2_{ih}\cdot y_{inh}
  \;\leq\; t_{nh}^{2}
  \qquad \forall\; n \in \M,\ h \in \H
\end{equation}
This is a second-order cone (SOC) constraint. Together, C$_{1a}$ + C$_{1b}$ guarantee:
\[
  P\!\bigl[\text{total demand on machine }n\text{ in period }h \leq \Ceff_n\bigr]
  \;\geq\; 1-\varepsilon.
\]

\paragraph{C\textsubscript{2} --- Priority Link.}
A shared tenant can receive allocation only on machines it is assigned to:
\begin{equation}\tag{C$_2$}
  f_{inh} \leq C_n \cdot y_{inh}
  \qquad \forall\; i \in \Ts,\ n \in \M,\ h \in \H
\end{equation}

\paragraph{C\textsubscript{3} --- Demand Satisfaction.}
Every shared tenant must receive at least its feedback-adjusted demand every period:
\begin{equation}\tag{C$_3$}
  \sum_{n \in \M} f_{inh} \geq \ufb_{ih}
  \qquad \forall\; i \in \Ts,\ h \in \H
\end{equation}

\paragraph{C\textsubscript{4} --- Fairness.}
$\sigma$ is the minimum demand-satisfaction fraction across all shared tenants:
\begin{equation}\tag{C$_4$}
  \sigma \;\leq\;
  \frac{\displaystyle\sum_{n \in \M,\,h \in \H} f_{inh}}
       {\displaystyle\sum_{h \in \H} \ufb_{ih}}
  \qquad \forall\; i \in \Ts
\end{equation}

\subsection{Mix Bonus Constraints}

$\mathrm{mix}_{nh}=1$ iff machine $n$ has at least one heavy \emph{and} one light tenant in period $h$.
This is a linearized AND gate:

\begin{align}
  \mathrm{hH}_{nh} &\geq y_{inh} & &\forall\; i\in\Th,\; n\in\M,\; h\in\H
    \tag{Mix$_{\mathrm{H}}$}\\[2pt]
  \mathrm{hL}_{nh} &\geq y_{jnh} & &\forall\; j\in\Tl,\; n\in\M,\; h\in\H
    \tag{Mix$_{\mathrm{L}}$}\\[2pt]
  \mathrm{mix}_{nh} &\leq \mathrm{hH}_{nh} & &\forall\; n\in\M,\; h\in\H
    \tag{Mix$_1$}\\[2pt]
  \mathrm{mix}_{nh} &\leq \mathrm{hL}_{nh} & &\forall\; n\in\M,\; h\in\H
    \tag{Mix$_2$}\\[2pt]
  \mathrm{mix}_{nh} &\geq \mathrm{hH}_{nh}+\mathrm{hL}_{nh}-1 & &\forall\; n\in\M,\; h\in\H
    \tag{Mix$_3$}
\end{align}

% ─────────────────────────────────────────────────────────────
\section{Objective}

\begin{equation}
\boxed{
\min\;
  \lambda_0\!\underbrace{\sum_{n\in\Mb}\!\pi_n z^{\mathrm{on}}_n}_{\text{InfraCost}}
  \;-\;
  \lambda_1\,\sigma
  \;-\;
  \lambda_2\!\underbrace{\sum_{n\in\M,\,h\in\H}\!\mathrm{mix}_{nh}}_{\text{MixTotal}}
}
\end{equation}

\begin{enumerate}[label=(\arabic*)]
  \item \textbf{InfraCost} (minimize): activate as few additional machines as possible.
  \item \textbf{Fairness} $\sigma$ (maximize): every shared tenant receives its fair share.
  \item \textbf{MixTotal} (maximize): reward co-locating heavy-demand and light-demand tenants.
\end{enumerate}

% ─────────────────────────────────────────────────────────────
\section{Uncertainty Model (Cantelli Bound)}

\subsection*{Why Cantelli?}
We know only mean $u_{ih}$ and variance $\sigma^2_{ih}$ of tenant demand — not the full distribution.
Cantelli's inequality is \emph{distribution-free}: it bounds the one-sided tail using only
mean and variance.

\subsection*{Applied to the Capacity Constraint}
Let $D_{nh}=\sum_{i\in\Ts}D_{inh}$ be the total random demand on machine $n$ in period $h$, where $D_{inh}$ has mean $f_{inh}$ and variance $\sigma^2_{ih}y_{inh}$.

By Cantelli's inequality:
\[
  P\!\Bigl[D_{nh} > \mu_{nh} + \kappa\,\sigma^{\mathrm{tot}}_{nh}\Bigr]
  \;\leq\; \frac{1}{1+\kappa^2} = \varepsilon,
\]
where $\mu_{nh}=\sum_i f_{inh}$ and $\left(\sigma^{\mathrm{tot}}_{nh}\right)^2=\sum_i\sigma^2_{ih}y_{inh}$.

Constraints C$_{1a}$+C$_{1b}$ together enforce $\mu_{nh}+\kappa\sigma^{\mathrm{tot}}_{nh}\leq\Ceff_n z_{nh}$, giving:
\[
  P\!\Bigl[D_{nh}\leq \Ceff_n z_{nh}\Bigr] \;\geq\; 1-\varepsilon.
\]
For $\varepsilon=0.10$, $\kappa=3.0$: machines overflow at most 10\% of the time
under the worst-case demand distribution with the given mean and variance.

% ─────────────────────────────────────────────────────────────
\section{Feedback Integration}

After the first planning horizon, the Real-Time layer reports $\bar{v}_n$ (SLA violation rate)
and $\bar{W}_i$ (average wait time per tenant), which update the plan parameters:

\begin{align}
  \Ceff_n &= C_n(1-\alpha\bar{v}_n)
    & &\text{(shrink capacity of overloaded machines)} \\[4pt]
  \ufb_{ih} &= u_{ih}\!\left(1+\beta\min\!\left(2,\,\frac{\bar{W}_i}{\bar{W}_{\mathrm{ref}}}\right)\right)
    & &\text{(inflate demand for tenants with long waits, capped at }2\times\text{)}
\end{align}

The cap at 2 prevents extreme demand inflation from making the MISOCP infeasible.
Default: $\alpha=0.5$,\ $\beta=0.3$,\ $\bar{W}_{\mathrm{ref}}=10$ intervals
(calibrated to the simulated time-step, not wall-clock seconds).

% ─────────────────────────────────────────────────────────────
\section{Connection to Real-Time Model}

The Cluster Manager uses the plan output as follows each simulation interval:
\begin{enumerate}
  \item Resolve $h = (\text{sim\_interval} \;\mathbin{/\!/}\; \texttt{period\_steps}) \bmod |\H|$.
  \item Get the group list for period $h$.
  \item For each group $g$:
    \begin{enumerate}[label=(\alph*)]
      \item Filter job queue: jobs with $\text{tenant\_id}\in g.\text{tenant\_ids}$.
      \item Filter nodes: machines with $\text{node\_id}\in g.\text{machine\_ids}$.
      \item Call $\mathrm{solve}(\text{group\_jobs},\,\text{group\_nodes},\,W_t,\,K,\,\text{time\_limit\_ms})$.
      \item Place returned assignments; unplaced jobs return to queue with a wait-time bump.
    \end{enumerate}
\end{enumerate}

The Real-Time model has \textbf{no knowledge} of tenants, plan-ahead, or machine assignments.
It receives only the pre-filtered jobs and nodes, and returns the best placement given capacity.
